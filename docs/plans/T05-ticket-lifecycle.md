# T05 — Ticket lifecycle (central vertical slice)

> **For agentic workers:** REQUIRED SUB-SKILL: use superpowers:subagent-driven-development or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax. Run subtasks in the mini-wave order below; parallel subtasks are run by separate subagents on disjoint files in the same worktree.

## 1. Header

- **Objective:** customers and staff can create, view, list (active / resolved), update, re-prioritize and move tickets through the status state machine, through both the Inertia UI and `/api/v1`, with SLA side effects, attachments and a complete activity timeline.
- **Depends on:** T02 (BusinessCalendar, SlaCalculator), T03 (TicketNumberGenerator, `config/tickets.php`), T04 (policies, `tests/Support/AuthorizationMatrix.php`).
- **Wave:** 3 (alone). Internal mini-waves: 5.1 → (5.2 ∥ 5.3) → (5.4 ∥ 5.5) → 5.6.
- **TDs implemented:** TD-001, TD-012, TD-023, TD-024 (ticket-level attachments), TD-026, TD-040, TD-042, TD-043, TD-053, TD-061 (storage half), TD-070 (all event classes), TD-101, TD-120, TD-130.

<?php

declare(strict_types=1);

namespace App\Services\Sla;

use App\Enums\SlaState;
use App\Enums\TicketStatus;
use Carbon\CarbonImmutable;
use Illuminate\Database\Eloquent\Builder;

/**
 * SQL twin of Ticket::slaState(). Every predicate leads with `status` so MySQL can use the
 * (status, sla_due_at) and (status, sla_warning_at) indexes (TD-021, TD-023).
 */
final class SlaQuery
{
    /**
     * @template TModel of \Illuminate\Database\Eloquent\Model
     *
     * @param  Builder<TModel>  $q
     * @return Builder<TModel>
     */
    public static function whereState(Builder $q, SlaState $state, CarbonImmutable $now): Builder
    {
        return match ($state) {
            SlaState::Stopped => $q->whereIn($q->qualifyColumn('status'), self::values(TicketStatus::finished())),
            SlaState::Paused => self::active($q)->whereNotNull($q->qualifyColumn('sla_paused_at')),
            SlaState::Breached => self::whereBreached($q, $now),
            SlaState::Approaching => self::whereApproaching($q, $now),
            SlaState::Within => self::running($q)->where($q->qualifyColumn('sla_warning_at'), '>', $now),
        };
    }

    /**
     * @template TModel of \Illuminate\Database\Eloquent\Model
     *
     * @param  Builder<TModel>  $q
     * @return Builder<TModel>
     */
    public static function whereBreached(Builder $q, CarbonImmutable $now): Builder
    {
        return self::running($q)->where($q->qualifyColumn('sla_due_at'), '<=', $now);
    }

    /**
     * @template TModel of \Illuminate\Database\Eloquent\Model
     *
     * @param  Builder<TModel>  $q
     * @return Builder<TModel>
     */
    public static function whereApproaching(Builder $q, CarbonImmutable $now): Builder
    {
        return self::running($q)
            ->where($q->qualifyColumn('sla_warning_at'), '<=', $now)
            ->where($q->qualifyColumn('sla_due_at'), '>', $now);
    }

    /**
     * @template TModel of \Illuminate\Database\Eloquent\Model
     *
     * @param  Builder<TModel>  $q
     * @return Builder<TModel>
     */
    private static function running(Builder $q): Builder
    {
        return self::active($q)->whereNull($q->qualifyColumn('sla_paused_at'));
    }

    /**
     * @template TModel of \Illuminate\Database\Eloquent\Model
     *
     * @param  Builder<TModel>  $q
     * @return Builder<TModel>
     */
    private static function active(Builder $q): Builder
    {
        return $q->whereIn($q->qualifyColumn('status'), self::values(TicketStatus::active()));
    }

    /**
     * @param  list<TicketStatus>  $statuses
     * @return list<string>
     */
    private static function values(array $statuses): array
    {
        return array_map(static fn (TicketStatus $s): string => $s->value, $statuses);
    }
}

<?php

declare(strict_types=1);

namespace App\Services\Attachments;

use App\Models\Attachment;
use App\Models\Ticket;
use App\Models\TicketReply;
use App\Models\User;
use Illuminate\Contracts\Config\Repository as Config;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Filesystem\FilesystemManager;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Str;
use RuntimeException;
use Throwable;

/**
 * Files cannot take part in a database transaction (TD-053), so every failure path deletes
 * what was already written. Callers wrap storeMany in their transaction and call deleteFiles
 * if the transaction fails after storeMany returned.
 */
final class AttachmentStorer
{
    public function __construct(
        private readonly FilesystemManager $storage,
        private readonly Config $config,
    ) {}

    /**
     * @param  list<UploadedFile>  $files
     * @return Collection<int, Attachment>
     */
    public function storeMany(Ticket $ticket, ?TicketReply $reply, User $uploader, array $files, bool $isInternal): Collection
    {
        /** @var Collection<int, Attachment> $attachments */
        $attachments = new Collection;

        if ($files === []) {
            return $attachments;
        }

        $disk = $this->disk();
        $directory = $this->config->get('attachments.directory', 'attachments').'/'.$ticket->id;
        $storedPaths = [];

        try {
            $pending = [];
            foreach ($files as $file) {
                $extension = strtolower($file->extension() ?: $file->getClientOriginalExtension() ?: 'bin');
                $path = $file->storeAs($directory, Str::uuid()->toString().'.'.$extension, ['disk' => $disk]);

                if ($path === false) {
                    throw new RuntimeException('Could not store attachment '.$file->getClientOriginalName());
                }

                $storedPaths[] = $path;
                $pending[] = [$file, $path];
            }

            foreach ($pending as [$file, $path]) {
                $attachment = new Attachment;
                $attachment->forceFill([
                    'ticket_id' => $ticket->id,
                    'reply_id' => $reply?->id,
                    'uploaded_by' => $uploader->id,
                    'disk' => $disk,
                    'path' => $path,
                    'original_name' => mb_substr($file->getClientOriginalName(), 0, 255),
                    'mime_type' => $file->getMimeType() ?? 'application/octet-stream',
                    'size_bytes' => (int) $file->getSize(),
                    'is_internal' => $isInternal,
                ])->save();

                $attachments->push($attachment);
            }
        } catch (Throwable $e) {
            $this->storage->disk($disk)->delete($storedPaths);

            throw $e;
        }

        return $attachments;
    }

    /**
     * @param  Collection<int, Attachment>  $attachments
     */
    public function deleteFiles(Collection $attachments): void
    {
        $attachments
            ->groupBy('disk')
            ->each(function (Collection $group, string $disk): void {
                $this->storage->disk($disk)->delete($group->pluck('path')->all());
            });
    }

    private function disk(): string
    {
        $disk = $this->config->get('attachments.disk', 'local');

        return is_string($disk) ? $disk : 'local';
    }
}

<?php

declare(strict_types=1);

namespace App\Actions\Tickets;

use App\Enums\TicketStatus;
use App\Events\TicketResolved;
use App\Events\TicketStatusChanged;
use App\Exceptions\Domain\InvalidStatusTransition;
use App\Models\Ticket;
use App\Models\User;
use App\Services\Sla\SlaCalculator;
use Carbon\CarbonImmutable;
use Illuminate\Support\Facades\DB;

final class ChangeTicketStatus
{
    public function __construct(
        private readonly SlaCalculator $sla,
    ) {}

    /**
     * A null actor means the system (e.g. a customer reply resuming a waiting ticket).
     */
    public function handle(?User $actor, Ticket $ticket, TicketStatus $to): Ticket
    {
        $now = CarbonImmutable::now();

        return DB::transaction(function () use ($actor, $ticket, $to, $now): Ticket {
            $this->lockAndRefresh($ticket);

            $from = $ticket->status;

            if (! $from->canTransitionTo($to)) {
                throw InvalidStatusTransition::between($from, $to);
            }

            if ($from === TicketStatus::WaitingForCustomer) {
                $this->sla->resume($ticket, $now);
            }

            if ($to === TicketStatus::WaitingForCustomer) {
                $this->sla->pause($ticket, $now);
            }

            if ($to === TicketStatus::Resolved) {
                $ticket->resolved_at = $now;
            }

            // Reopening keeps the original SLA deadline; only the resolution stamp is withdrawn (TD-042).
            if ($from === TicketStatus::Resolved && $to === TicketStatus::InProgress) {
                $ticket->resolved_at = null;
            }

            if ($to === TicketStatus::Closed) {
                $ticket->closed_at = $now;
            }

            $ticket->status = $to;
            $ticket->save();

            TicketStatusChanged::dispatch($ticket, $actor, $from, $to);

            if ($to === TicketStatus::Resolved) {
                TicketResolved::dispatch($ticket, $actor);
            }

            return $ticket;
        });
    }

    /**
     * The caller's instance may be stale; the transition must be decided on the locked row.
     */
    private function lockAndRefresh(Ticket $ticket): void
    {
        $locked = Ticket::query()->whereKey($ticket->getKey())->lockForUpdate()->firstOrFail();
        $ticket->setRawAttributes($locked->getAttributes(), true);
    }
}

use App\Enums\TicketPriority;

        return DB::transaction(function () use ($actor, $ticket, $to, $now): Ticket {

            return $ticket;
        });

<?php

declare(strict_types=1);

namespace App\Actions\Tickets;

use App\Events\TicketUpdated;
use App\Models\Ticket;
use App\Models\User;

final class UpdateTicket
{
    /** Only content fields; status, priority, assignment and SLA have their own Actions. */
    private const FIELDS = ['subject', 'description', 'category_id'];

    /**
     * Single-row write with no read-then-decide step, so no transaction is needed (TD-053).
     *
     * @param  array<string, mixed>  $data
     */
    public function handle(User $actor, Ticket $ticket, array $data): Ticket
    {
        $changes = [];

        foreach (self::FIELDS as $field) {
            if (! array_key_exists($field, $data)) {
                continue;
            }

            $old = $ticket->getAttribute($field);
            $new = $field === 'category_id' ? (int) $data[$field] : (string) $data[$field];
            $oldNormalized = $field === 'category_id' ? (int) $old : (string) $old;

            if ($oldNormalized === $new) {
                continue;
            }

            $changes[$field] = ['from' => $oldNormalized, 'to' => $new];
        }

        if ($changes === []) {
            return $ticket;
        }

        $ticket->fill(array_map(static fn (array $change): mixed => $change['to'], $changes));
        $ticket->save();

        TicketUpdated::dispatch($ticket, $actor, $changes);

        return $ticket;
    }
}

<?php

declare(strict_types=1);

namespace App\Actions\Tickets;

use App\Enums\TicketPriority;
use App\Enums\TicketStatus;
use App\Events\TicketCreated;
use App\Models\Attachment;
use App\Models\Ticket;
use App\Models\User;
use App\Services\Attachments\AttachmentStorer;
use App\Services\Sla\SlaCalculator;
use App\Services\Tickets\TicketNumberGenerator;
use Carbon\CarbonImmutable;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use Throwable;

final class CreateTicket
{
    public function __construct(
        private readonly TicketNumberGenerator $numbers,
        private readonly SlaCalculator $sla,
        private readonly AttachmentStorer $attachments,
    ) {}

    /**
     * @param  array{subject: string, description: string, priority: string, category_id: int|string, customer_id?: int|string|null}  $data
     * @param  list<UploadedFile>  $files
     */
    public function handle(User $actor, array $data, array $files = []): Ticket
    {
        $now = CarbonImmutable::now();

        /** @var Collection<int, Attachment> $stored */
        $stored = new Collection;

        try {
            return DB::transaction(function () use ($actor, $data, $files, $now, &$stored): Ticket {
                $ticket = new Ticket([
                    'subject' => $data['subject'],
                    'description' => $data['description'],
                    'priority' => TicketPriority::from($data['priority']),
                    'category_id' => (int) $data['category_id'],
                    'customer_id' => $this->customerId($actor, $data),
                    'created_by_id' => $actor->id,
                ]);
                // The number shares this transaction, so a rollback also rolls back the counter (TD-022).
                $ticket->number = $this->numbers->next($now);
                $ticket->status = TicketStatus::Open;
                $this->sla->applyInitial($ticket, $now);
                $ticket->save();
                // Loads database defaults (escalation_level) so callers and resources see the stored row.
                $ticket->refresh();

                $stored = $this->attachments->storeMany($ticket, null, $actor, $files, false);

                TicketCreated::dispatch($ticket, $actor);

                return $ticket;
            });
        } catch (Throwable $e) {
            $this->attachments->deleteFiles($stored);

            throw $e;
        }
    }

    /**
     * @param  array<string, mixed>  $data
     */
    private function customerId(User $actor, array $data): int
    {
        $requested = $data['customer_id'] ?? null;

        if ($requested !== null && $actor->can('createOnBehalf', Ticket::class)) {
            return (int) $requested;
        }

        return $actor->id;
    }
}

<?php

declare(strict_types=1);

namespace App\Listeners;

use App\Enums\ActivityType;
use App\Enums\AssignmentMethod;
use App\Enums\EscalationLevel;
use App\Enums\TicketStatus;
use App\Events\TicketAssigned;
use App\Events\TicketCreated;
use App\Events\TicketPriorityChanged;
use App\Events\TicketReplied;
use App\Events\TicketSlaApproaching;
use App\Events\TicketSlaBreached;
use App\Events\TicketStatusChanged;
use App\Events\TicketUpdated;
use App\Models\Category;
use App\Models\Ticket;
use App\Models\User;
use Throwable;

/**
 * The only writer of the ticket timeline (TD-026). TicketResolved is deliberately not handled:
 * the matching status_changed entry already records the resolution.
 */
final class RecordTicketActivity
{
    public function handleTicketCreated(TicketCreated $event): void
    {
        $ticket = $event->ticket;

        if ($event->imported) {
            $this->record($ticket, $event->actor, ActivityType::Imported, [
                'number' => $ticket->number,
                'external_reference' => $ticket->external_reference,
            ]);

            return;
        }

        $this->record($ticket, $event->actor, ActivityType::Created, [
            'number' => $ticket->number,
            'priority' => $ticket->priority->value,
            'category_id' => $ticket->category_id,
        ]);
    }

    public function handleTicketUpdated(TicketUpdated $event): void
    {
        $changes = $event->changes;

        if (array_key_exists('category_id', $changes)) {
            $from = (int) $changes['category_id']['from'];
            $to = (int) $changes['category_id']['to'];
            $names = Category::query()->whereKey([$from, $to])->pluck('name', 'id');

            $this->record($event->ticket, $event->actor, ActivityType::CategoryChanged, [
                'from' => ['id' => $from, 'name' => $names->get($from)],
                'to' => ['id' => $to, 'name' => $names->get($to)],
            ]);

            unset($changes['category_id']);
        }

        if ($changes === []) {
            return;
        }

        // Descriptions can be long and are not useful in a one-line timeline; only the subject keeps its values.
        $properties = ['fields' => array_keys($changes), 'from' => [], 'to' => []];
        if (array_key_exists('subject', $changes)) {
            $properties['from']['subject'] = $changes['subject']['from'];
            $properties['to']['subject'] = $changes['subject']['to'];
        }

        $this->record($event->ticket, $event->actor, ActivityType::Updated, $properties);
    }

    public function handleTicketAssigned(TicketAssigned $event): void
    {
        $type = $event->method === AssignmentMethod::Claimed ? ActivityType::Claimed : ActivityType::Assigned;

        $this->record($event->ticket, $event->actor, $type, [
            'method' => $event->method->value,
            'from' => $event->previousAgent === null ? null : $this->summary($event->previousAgent),
            'to' => $this->summary($event->agent),
        ]);
    }

    public function handleTicketStatusChanged(TicketStatusChanged $event): void
    {
        $ticket = $event->ticket;

        $this->record($ticket, $event->actor, ActivityType::StatusChanged, [
            'from' => $event->from->value,
            'to' => $event->to->value,
        ]);

        if ($event->to === TicketStatus::WaitingForCustomer) {
            $this->record($ticket, $event->actor, ActivityType::SlaPaused, [
                'paused_at' => $ticket->sla_paused_at?->toIso8601String(),
            ]);
        }

        if ($event->from === TicketStatus::WaitingForCustomer) {
            $this->record($ticket, $event->actor, ActivityType::SlaResumed, [
                'due_at' => $ticket->sla_due_at?->toIso8601String(),
                'warning_at' => $ticket->sla_warning_at?->toIso8601String(),
            ]);
        }
    }

    public function handleTicketPriorityChanged(TicketPriorityChanged $event): void
    {
        $this->record($event->ticket, $event->actor, ActivityType::PriorityChanged, [
            'from' => $event->from->value,
            'to' => $event->to->value,
            'due_at' => $event->ticket->sla_due_at?->toIso8601String(),
        ]);
    }

    public function handleTicketReplied(TicketReplied $event): void
    {
        $type = $event->reply->is_internal ? ActivityType::InternalNote : ActivityType::Replied;

        $this->record($event->ticket, $event->author, $type, [
            'reply_id' => $event->reply->id,
        ]);
    }

    public function handleTicketSlaApproaching(TicketSlaApproaching $event): void
    {
        $this->record($event->ticket, null, ActivityType::SlaApproaching, [
            'due_at' => $event->ticket->sla_due_at?->toIso8601String(),
        ]);
    }

    public function handleTicketSlaBreached(TicketSlaBreached $event): void
    {
        $this->record($event->ticket, null, ActivityType::Escalated, [
            'level' => EscalationLevel::Breached->value,
            'due_at' => $event->ticket->sla_due_at?->toIso8601String(),
        ]);
    }

    /**
     * @param  array<string, mixed>  $properties
     */
    private function record(Ticket $ticket, ?User $actor, ActivityType $type, array $properties): void
    {
        try {
            // performedOn() first: it initialises the pending activity that causedByAnonymous() edits.
            $logger = activity('ticket')
                ->performedOn($ticket)
                ->event($type->value)
                ->withProperties($properties);

            // Without causedByAnonymous() the package would default the causer to the logged-in user,
            // which would misattribute system transitions made during a request.
            $actor === null ? $logger->causedByAnonymous() : $logger->causedBy($actor);

            $logger->log($type->describe($properties));
        } catch (Throwable $e) {
            // The business change is already committed; a missing timeline entry must not fail it (TD-026).
            report($e);
        }
    }

    /**
     * @return array{id: int, name: string}
     */
    private function summary(User $user): array
    {
        return ['id' => $user->id, 'name' => $user->name];
    }
}

                'paused_at' => $this->sla_paused_at?->toIso8601String(),
            ],
            'escalation_level' => $this->escalation_level->value,
            'created_at' => $this->created_at?->toIso8601String(),
            'updated_at' => $this->updated_at?->toIso8601String(),
            'resolved_at' => $this->resolved_at?->toIso8601String(),
            'closed_at' => $this->closed_at?->toIso8601String(),
            'can' => $this->when($user instanceof User, fn (): array => self::abilities($user, $this->resource)),
        ];
    }

    /**
     * The TicketCan shape (README §3.13). Every flag comes from TicketPolicy, so the UI only
     * mirrors what the server enforces.
     *
     * @return array{update: bool, changeStatus: bool, changePriority: bool, claim: bool, assign: bool, reply: bool, addInternalNote: bool, allowedStatuses: list<string>}
     */
    public static function abilities(User $user, Ticket $ticket): array
    {
        $allowed = array_values(array_filter(
            $ticket->status->allowedTransitions(),
            static fn (TicketStatus $to): bool => $user->can('changeStatus', [$ticket, $to]),
        ));

        return [
            'update' => $user->can('update', $ticket),
            'changeStatus' => $allowed !== [],
            'changePriority' => $user->can('changePriority', $ticket),
            // The policy ignores the assignee on purpose (a lost race must reach the Action and get 409);
            // the UI offers "claim" only for tickets that are still unassigned.
            'claim' => $user->can('claim', $ticket) && $ticket->assigned_agent_id === null,
            'assign' => $user->can('assign', $ticket),
            'reply' => $user->can('reply', $ticket),
            'addInternalNote' => $user->can('addInternalNote', $ticket),
            'allowedStatuses' => array_map(static fn (TicketStatus $s): string => $s->value, $allowed),
        ];
    }
}

use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Database\Eloquent\Builder;

/**
 * The single ticket-list query used by the web and API index endpoints. T08 plugs TicketFilter
 * into applyFilters(); visibility, scope, eager loading and pagination stay here.
 */
final class TicketListQuery
{
    private const LIST_COLUMNS = [
        'tickets.id', 'tickets.number', 'tickets.subject', 'tickets.status', 'tickets.priority',
        'tickets.category_id', 'tickets.customer_id', 'tickets.assigned_agent_id',
        'tickets.sla_started_at', 'tickets.sla_due_at', 'tickets.sla_warning_at', 'tickets.sla_paused_at',
        'tickets.escalation_level', 'tickets.resolved_at', 'tickets.closed_at',
        'tickets.created_at', 'tickets.updated_at',
    ];

    public function __construct(
        private readonly Config $config,
    ) {}

    /**
     * @param  array<string, mixed>  $validated
     * @return LengthAwarePaginator<int, Ticket>
     */
    public function paginate(User $viewer, array $validated, CarbonImmutable $now): LengthAwarePaginator
    {
        $statuses = ($validated['scope'] ?? 'active') === 'resolved' ? TicketStatus::finished() : TicketStatus::active();

        $query = Ticket::query()
            ->select(self::LIST_COLUMNS)
            ->visibleTo($viewer)
            ->whereIn('tickets.status', array_map(static fn (TicketStatus $s): string => $s->value, $statuses))
            ->with(['category:id,name', 'customer:id,name', 'assignedAgent:id,name'])
            // Counting only visible replies: a raw count would reveal internal notes to customers.
            ->withCount(['replies' => static fn (Builder $q) => $q->visibleTo($viewer)]);

        $query = $this->applyFilters($query, $validated, $viewer, $now);

        return $query->paginate($this->perPage($validated))->withQueryString();
    }

    /**
     * T05 only applies the default order. T08 replaces this body with
     * `return $this->filter->apply($query, $validated, $viewer, $now);`.
     *
     * @param  Builder<Ticket>  $query
     * @param  array<string, mixed>  $validated
     * @return Builder<Ticket>
     */
    private function applyFilters(Builder $query, array $validated, User $viewer, CarbonImmutable $now): Builder
    {
        return $query->orderByDesc('tickets.created_at')->orderByDesc('tickets.id');
    }

    /**
     * @param  array<string, mixed>  $validated
     */
    private function perPage(array $validated): int
    {
        $default = (int) $this->config->get('tickets.per_page.default', 20);
        $max = (int) $this->config->get('tickets.per_page.max', 100);
        $requested = isset($validated['per_page']) ? (int) $validated['per_page'] : $default;

        return max(1, min($requested, $max));
    }
}

                ->orderBy('id'),

<?php

declare(strict_types=1);

namespace App\Http\Requests\Tickets;

use App\Enums\Role;
use App\Enums\TicketPriority;
use App\Models\Ticket;
use App\Models\User;
use Closure;
use Illuminate\Auth\Access\Response;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Gate;
use Illuminate\Validation\Rule;
use Illuminate\Validation\Rules\File;

final class StoreTicketRequest extends FormRequest
{
    public function authorize(): Response
    {
        return Gate::forUser($this->user())->inspect('create', Ticket::class);
    }

    /**
     * @return array<string, mixed>
     */
    public function rules(): array
    {
        /** @var list<string> $extensions */
        $extensions = config('attachments.extensions', []);

        return [
            'subject' => ['required', 'string', 'max:255'],
            'description' => ['required', 'string', 'max:20000'],
            'priority' => ['required', Rule::enum(TicketPriority::class)],
            'category_id' => ['required', 'integer', Rule::exists('categories', 'id')->where('is_active', true)],
            // Staff identify the customer by email (unique, indexed); customers always create for themselves.
            'customer_email' => $this->canCreateOnBehalf()
                ? ['required', 'string', 'email', $this->mustBeCustomer()]
                : ['prohibited'],
            'attachments' => ['sometimes', 'array', 'max:'.(int) config('attachments.max_files', 5)],
            'attachments.*' => [File::types($extensions)->max((int) config('attachments.max_kb', 10240))],
        ];
    }

    /**
     * The $data array for CreateTicket::handle().
     *
     * @return array{subject: string, description: string, priority: string, category_id: int, customer_id?: int}
     */
    public function actionData(): array
    {
        $data = [
            'subject' => (string) $this->validated('subject'),
            'description' => (string) $this->validated('description'),
            'priority' => (string) $this->validated('priority'),
            'category_id' => (int) $this->validated('category_id'),
        ];

        $email = $this->validated('customer_email');
        if (is_string($email)) {
            $data['customer_id'] = (int) User::query()->where('email', $email)->value('id');
        }

        return $data;
    }

    /**
     * @return list<UploadedFile>
     */
    public function attachmentFiles(): array
    {
        $files = $this->file('attachments', []);

        return array_values(array_filter(
            is_array($files) ? $files : [$files],
            static fn (mixed $f): bool => $f instanceof UploadedFile,
        ));
    }

    private function canCreateOnBehalf(): bool
    {
        $user = $this->user();

        return $user instanceof User && $user->can('createOnBehalf', Ticket::class);
    }

    private function mustBeCustomer(): Closure
    {
        return static function (string $attribute, mixed $value, Closure $fail): void {
            $isCustomer = User::query()
                ->where('email', $value)
                ->role(Role::Customer->value)
                ->exists();

            if (! $isCustomer) {
                $fail('The selected customer does not exist.');
            }
        };
    }

        private readonly TicketPageData $pages,

    public function index(IndexTicketsRequest $request, TicketListQuery $tickets): InertiaResponse
    {
        $validated = $request->validated();

        return Inertia::render('tickets/Index', [
            'tickets' => TicketListResource::collection($tickets->paginate($this->user($request), $validated, CarbonImmutable::now())),
            'scope' => $request->scope(),
            'filters' => (object) $validated,
        ]);
    }

    public function create(Request $request): InertiaResponse
    {
        Gate::authorize('create', Ticket::class);

        return Inertia::render('tickets/Create', [
            'options' => $this->pages->formOptions(),
            'canCreateOnBehalf' => $this->user($request)->can('createOnBehalf', Ticket::class),
        ]);
    }

    public function store(StoreTicketRequest $request, CreateTicket $createTicket): RedirectResponse
    {
        $ticket = $createTicket->handle($this->user($request), $request->actionData(), $request->attachmentFiles());

        return to_route('tickets.show', $ticket)->with('success', "Ticket {$ticket->number} created.");
    }

    public function show(Request $request, Ticket $ticket): InertiaResponse
    {
        Gate::authorize('view', $ticket);

        $user = $this->user($request);
        $this->pages->loadForShow($ticket, $user);
        $can = TicketResource::abilities($user, $ticket);

        return Inertia::render('tickets/Show', [
            'ticket' => TicketResource::make($ticket)->resolve($request),
            'replies' => $this->replyItems($ticket, $request),
            'attachments' => AttachmentResource::collection($ticket->attachments)->resolve($request),
            'activities' => ActivityResource::collection($this->pages->activities($ticket, $user))->resolve($request),
            'agents' => $can['assign'] ? $this->pages->agentOptions() : [],
            'can' => $can,
            'options' => $this->pages->formOptions(),
        ]);
    }

    public function update(UpdateTicketRequest $request, Ticket $ticket, UpdateTicket $updateTicket): RedirectResponse
    {
        $updateTicket->handle($this->user($request), $ticket, $request->validated());

        return back()->with('success', 'Ticket updated.');
    }

    public function status(ChangeTicketStatusRequest $request, Ticket $ticket, ChangeTicketStatus $changeStatus): RedirectResponse
    {
        $ticket = $changeStatus->handle($this->user($request), $ticket, $request->targetStatus());

        return back()->with('success', "Status changed to {$ticket->status->label()}.");
    }

    public function priority(ChangeTicketPriorityRequest $request, Ticket $ticket, ChangeTicketPriority $changePriority): RedirectResponse
    {
        $ticket = $changePriority->handle($this->user($request), $ticket, $request->targetPriority());

        return back()->with('success', "Priority changed to {$ticket->priority->label()}.");
    }

    /**
     * The ReplyItem shape (resources/js/types/tickets.ts). T07's TicketReplyResource emits the same shape for the API.
     *
     * @return list<array<string, mixed>>
     */
    private function replyItems(Ticket $ticket, Request $request): array
    {
        return $ticket->replies
            ->map(static fn (TicketReply $reply): array => [
                'id' => $reply->id,
                'body' => $reply->body,
                'is_internal' => (bool) $reply->is_internal,
                'author' => ['id' => $reply->author->id, 'name' => $reply->author->name],
                'attachments' => AttachmentResource::collection($reply->attachments)->resolve($request),
                'created_at' => $reply->created_at?->toIso8601String(),
            ])
            ->values()
            ->all();
    }

    private function user(Request $request): User
    {
        $user = $request->user();
        assert($user instanceof User);

        return $user;
    }

**Layout (Inertia 3 kit).** `resources/js/app.ts` sets a default layout in `createInertiaApp({ layout })`: every page that isn't `Welcome`, `auth/*` or `settings/*` is already rendered inside `AppLayout`. Pages therefore **don't** wrap themselves in `<AppLayout>`, because that would render two sidebars. They pass the layout's `breadcrumbs` prop instead: as a static object in `defineOptions({ layout: { breadcrumbs } })` (the kit's `pages/Dashboard.vue` pattern), or with `setLayoutProps()` from `@inertiajs/vue3` when the value depends on page props (the kit's `pages/auth/TwoFactorChallenge.vue` pattern).         Gate::authorize('view', $ticket);

        return TicketResource::make($this->pages->loadSummary($ticket));
    }

    {
        $ticket = $updateTicket->handle($this->user($request), $ticket, $request->validated());

        return TicketResource::make($this->pages->loadSummary($ticket));
    }

    public function status(ChangeTicketStatusRequest $request, Ticket $ticket, ChangeTicketStatus $changeStatus): TicketResource
    {
        $ticket = $changeStatus->handle($this->user($request), $ticket, $request->targetStatus());

        return TicketResource::make($this->pages->loadSummary($ticket));
    }

### 5.6 Cross-cutting tests: authorization matrix and query counts

**Files:** `tests/Feature/Authorization/TicketRoutesAuthorizationTest.php`, `tests/Feature/Performance/TicketQueryCountTest.php`. Both are characterization tests over 5.4/5.5: they must pass on first run; to prove they bite, step 3 and step 7 temporarily break the code and expect FAIL, then revert.

The matrix uses T04's helper: `$matrix->actingAsRole(Role)` creates and authenticates a user of that role (web session and Sanctum) and returns it; `$matrix->expectAccess($method, $uri, $status)` sends the request (JSON for `/api/` URIs) with no payload and asserts the status. Because FormRequest authorization runs **before** validation, an empty payload separates the cases cleanly: forbidden → 403 / 404, allowed → validation 302 (web) or 422 (API), or success.

        ['admin', 'other', 'PATCH', '/status', 302, 422],

```php
<?php

declare(strict_types=1);

use App\Enums\TicketStatus;
use App\Events\TicketReplied;
use App\Models\Attachment;
use App\Models\Ticket;
use App\Models\TicketReply;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Laravel\Sanctum\Sanctum;

function ticketQueryCount(callable $request): int
{
    DB::flushQueryLog();
    DB::enableQueryLog();
    $request();
    $count = count(DB::getQueryLog());
    DB::disableQueryLog();

    return $count;
}

function ticketWithConversation(User $agent, int $replies): Ticket
{
    $ticket = Ticket::factory()->status(TicketStatus::InProgress)->assignedTo($agent)->create();

    for ($i = 0; $i < $replies; $i++) {
        $reply = TicketReply::factory()->create(['ticket_id' => $ticket->id, 'author_id' => User::factory()->agent()->create()->id, 'is_internal' => $i % 2 === 1]);
        Attachment::factory()->create(['ticket_id' => $ticket->id, 'reply_id' => $reply->id, 'is_internal' => (bool) $reply->is_internal]);
        TicketReplied::dispatch($ticket, $reply, $agent);
    }

    return $ticket;
}

beforeEach(function (): void {
    $this->admin = User::factory()->admin()->create();
    $this->agent = User::factory()->agent()->create();
});

it('renders the web ticket index with a constant number of queries', function (): void {
    $this->actingAs($this->admin);
    Ticket::factory()->status(TicketStatus::Open)->count(2)->create();
    $this->get('/tickets')->assertOk();

    $small = ticketQueryCount(fn () => $this->get('/tickets')->assertOk());

    // Every new ticket gets its own category, customer and agent, which is what an N+1 would multiply.
    Ticket::factory()->status(TicketStatus::Open)->count(15)->sequence(fn () => ['assigned_agent_id' => User::factory()->agent()->create()->id])->create();
    $large = ticketQueryCount(fn () => $this->get('/tickets')->assertOk());

    expect($large)->toBe($small)
        ->and($small)->toBeLessThanOrEqual(12);
});

it('serves the API ticket index with a constant number of queries', function (): void {
    Sanctum::actingAs($this->admin);
    Ticket::factory()->status(TicketStatus::Open)->count(2)->create();
    $this->getJson('/api/v1/tickets')->assertOk();

    $small = ticketQueryCount(fn () => $this->getJson('/api/v1/tickets')->assertOk());
    Ticket::factory()->status(TicketStatus::Open)->count(15)->create();
    $large = ticketQueryCount(fn () => $this->getJson('/api/v1/tickets')->assertOk());

    expect($large)->toBe($small);
});

it('renders the ticket show page with a constant number of queries', function (): void {
    $this->actingAs($this->admin);
    $small = ticketWithConversation($this->agent, 2);
    $large = ticketWithConversation($this->agent, 12);
    $this->get('/tickets/'.$small->number)->assertOk();

    $smallCount = ticketQueryCount(fn () => $this->get('/tickets/'.$small->number)->assertOk());
    $largeCount = ticketQueryCount(fn () => $this->get('/tickets/'.$large->number)->assertOk());

    expect($largeCount)->toBe($smallCount)
        ->and($smallCount)->toBeLessThanOrEqual(20);
});
```
